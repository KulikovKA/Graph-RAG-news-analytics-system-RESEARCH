```cypher
WITH 'relex_corpus100_stage_3a5_11_v1' AS ns
UNWIND ['RELEX_MENTIONS', 'RELEX_REL', 'SIMILAR_TO', 'HAS_SHARED_CONCEPT'] AS kind
CALL (ns, kind) {
  MATCH p = (a {namespace: ns})-[r {namespace: ns}]->(b {namespace: ns})
  WHERE type(r) = kind
  RETURN p LIMIT 40
}
RETURN p
```
